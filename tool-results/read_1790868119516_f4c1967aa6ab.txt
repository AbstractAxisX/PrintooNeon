   105→function useCountdown(endsAt) {
   106→  const [left, setLeft] = useState(() => Math.max(0, endsAt - Date.now()));
   107→  useEffect(() => {
   108→    const iv = setInterval(() => setLeft(Math.max(0, endsAt - Date.now())), 1000);
   109→    return () => clearInterval(iv);
   110→  }, [endsAt]);
   111→  const d = Math.floor(left / 864e5), h = Math.floor(left / 36e5) % 24,
   112→        m = Math.floor(left / 6e4) % 60, s = Math.floor(left / 1e3) % 60;
   113→  return { d, h, m, s };
   114→}

   117→function flyToCart(fromEl) {
   118→  try {
   119→    const cart = document.getElementById('cartBtn');
   120→    if (!cart || !fromEl) return;

   122→    const dot = document.createElement('div');
   123→    dot.className = 'fly-dot';
   124→    dot.style.left = (a.left + a.width / 2) + 'px';
   125→    dot.style.top = (a.top + a.height / 2) + 'px';
   126→    document.body.appendChild(dot);
   127→    requestAnimationFrame(() => requestAnimationFrame(() => {
   128→      dot.style.transform = 'translate(' + ((b.left + b.width / 2) - (a.left + a.width / 2)) + 'px,' +
   129→                                          ((b.top + b.height / 2) - (a.top + a.height / 2)) + 'px) scale(.35)';
   130→      dot.style.opacity = '0';
   131→    }));
   132→    setTimeout(() => dot.remove(), 650);
   133→  } catch (err) { /* بی‌خیال */ }
   134→}

   136→/* ---------- هوک Embla (در پروژه‌ی ری‌ال جایگزین useEmblaCarousel رسمی می‌شود) ---------- */
   137→function useEmbla(options) {
   138→  const ref = useRef(null);
   139→  const [api, setApi] = useState(null);
   140→  useEffect(() => {
   141→    const el = ref.current;
   142→    if (!el) return undefined;
   143→    if (!window.EmblaCarousel) { el.classList.add('embla--native'); return undefined; }
   144→    const inst = window.EmblaCarousel(el, options);
   145→    setApi(inst);
   146→    return () => { inst.destroy(); setApi(null); };
   147→  }, []);
   148→  return [ref, api];
   149→}

   550→/* ---------- کروسل: درگ + اسکرول لمسی، بدون دکمه‌ی جهت ---------- */
   551→function Carousel({ children, tone, label }) {
   552→  const t = toneStyles(tone);
   553→  const [vpRef, embla] = useEmbla({ axis: 'x', direction: 'rtl', align: 'start', containScroll: 'trimSnaps' });
   554→  const [prog, setProg] = useState(0);
   555→  const down = useRef(false);
   556→  const moved = useRef(false);
   557→
   558→  useEffect(() => {
   559→    if (!embla) return undefined;
   560→    const upd = () => { const pr = embla.scrollProgress(); setProg(pr < 0 ? 0 : pr > 1 ? 1 : pr); };
   561→    const onDown = () => { down.current = true; moved.current = false; };
   562→    const onUp = () => { down.current = false; };
   563→    const onScroll = () => { upd(); if (down.current) moved.current = true; };
   564→    upd();
   565→    embla.on('scroll', onScroll);
   566→    embla.on('reInit', upd);
   567→    embla.on('pointerDown', onDown);
   568→    embla.on('pointerUp', onUp);
   569→    return () => {
   570→      embla.off('scroll', onScroll); embla.off('reInit', upd);
   571→      embla.off('pointerDown', onDown); embla.off('pointerUp', onUp);
   572→    };
   573→  }, [embla]);
   574→
   575→  /* اگر کارت درگ شده باشد، کلیکِ ناخواسته‌ی بعد از رها کردن حذف می‌شود */
   576→  const onClickCapture = e => {
   577→    if (moved.current) { e.preventDefault(); e.stopPropagation(); }
   578→  };
   579→
   580→  /* کلیک/لمس روی نوار پیشرفت = پرش به همان نقطه (جایگزین دکمه‌ها) */
   581→  const seek = e => {
   582→    if (!embla) return;
   583→    const r = e.currentTarget.getBoundingClientRect();
   584→    const ratio = Math.min(1, Math.max(0, (r.width - (e.clientX - r.left)) / r.width)); /* RTL */
   585→    const last = embla.scrollSnapList().length - 1;
   586→    if (last > 0) embla.scrollTo(Math.round(ratio * last));
   587→  };
   588→
   589→  const onKey = e => {
   590→    if (!embla) return;
   591→    if (e.key === 'ArrowLeft') embla.scrollNext();
   592→    if (e.key === 'ArrowRight') embla.scrollPrev();
   593→  };
   594→
   595→  /* toArray: فرزندهای null حذف می‌شوند تا اسلات خالی ساخته نشود */
   596→  const slides = React.Children.toArray(children);
   597→
   598→  return (
   599→    <div className="relative">
   600→      <div ref={vpRef} className="embla__viewport" onClickCapture={onClickCapture}
   601→        role="region" aria-roledescription="کروسل" aria-label={label}>
   602→        <div className="embla__container">
   603→          {slides.map((child, i) => (
   604→            <div className="embla__slide" key={i}
   605→              role="group" aria-roledescription="اسلاید"
   606→              aria-label={fa(i + 1) + ' از ' + fa(slides.length)}>
   607→              {child}
   608→            </div>
   609→          ))}
   610→        </div>
   611→      </div>
   612→
   613→      <div className="flex items-center gap-3 mt-1">
   614→        <span className={'shrink-0 hidden sm:flex items-center gap-1 text-[10px] font-medium ' + t.meta}>
   615→          <svg className="w-[13px] h-[13px]" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="1.8" strokeLinecap="round" strokeLinejoin="round"><path d="M8 8l-4 4 4 4M16 8l4 4-4 4M4.5 12h15"/></svg>
   616→          کشیدن
   617→        </span>
   618→        <div className="flex-1 py-1.5 cursor-pointer" role="slider" tabIndex={0}
